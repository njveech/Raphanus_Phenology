# load the data
clim <- read.csv("Output_Files/climateNA_full_data_tall.csv", header = T)
specimens <- read.csv("Input_Files/FullSpecimenCCH2Data.csv", header = T)

#Note: Will eventually need to combine back in at least phenology scoring data(maybe not coords, etc from CCH2). This could likely best be done using the full entire class dataset from earlier code that is within Raphanus Phenology markdown


#subset to keep just the columns we need 
specimens <- specimens %>% dplyr::select(id, year, month, day)


#WUsing the full year (12 months) of data starting with the day and month of collection, but in the previous year, and ending with the date of collection. 

# METHODS: First, we need to calculate the date range. We will only include the month of collection in the year of collection if it was collected after the first week, so on the 8th or later. If it was collected before the 8th (1-7th), we do NOT include that month for the year of collection, and instead begin from the full month completed prior to the specimen's collection Additionally, if a specimen did not have a specific date of collection listed (n = 17), whcih displays as day=0 in the data frame, we do not feel confident in assuming that we should use said listed month as the end date, so the specimen will be treated the same as a specimen collected on the 1st-7th of the month. 


# Step 1: Identify the start and end month for the "climate year" for each specimen within the specimen df.These new columns should be in the clim_date format with year and month together (e.g. 202412 for December 2024).

# Step 1a: Identify the end month (this will then define the start month as 11 mo before). The newly coded end_mo column should contain the month of collection ( = month) if the day of collection column has a value >7. Otherwise, the end_mo column is assigned the month previous to collection (month -1). Special care was taken to address the cyclical nature of the calendar year, where Janaury is preceded by December.

specimens <- specimens %>% mutate(
  end_mo = if_else(day > 7, month, 
if_else(month == 1, 12, month - 1))) 

# Step 1b: Identify the start month and create its column; this will simply be the end month + 1 in order to capture the full 12 month (1 year) cycle. Once again, special care was taken to address the cyclical nature of the calendar year, where Janaury is preceded by December.


specimens <- specimens %>% mutate(start_mo = if_else(end_mo != 12, end_mo + 1, 1))

# Step 1c: Now there are two new columns in the specimens df, one each for start and end month. The year of collection must be appended to the month in order to get the start date and end date pertaining to each specimen that will narrow down the ClimateNA data. Special precautions were included to address the End Date scenario where a specimen collected before January 8th is assigned December as an end date, but the year must be listed (for both start and end date), as the year prior. Additionally, when constructing the Start Date character string, special conditions were added for specimens collected past December 7th, where, unlike any other collection date, the Start and End Date fall in the same year. 


#For End Date:
specimens <- specimens %>% mutate(end_date = if_else(end_mo == 12 & month == 1, paste0(year-1, str_pad(end_mo, 2, pad = "0")), paste0(year, str_pad(end_mo, 2, pad = "0"))), end_mo = as.integer(end_mo))

#For Start Date:
specimens <- specimens %>% mutate(start_date = if_else(month == 12 & end_mo == 12, paste0(year, str_pad(start_mo, 2, pad = "0")), paste0(year -1, str_pad(start_mo, 2, pad = "0"))), start_mo = as.integer(start_mo))

# Step 2: Join the specimens and clim dataframes to combine the newly adjusted CCH2 Data with the unfiltered ClimateNA data frame.

big_comb <- left_join(clim, specimens, by = "id")

# Step 3: To subset the climate data, keep only the rows for each specimen ID where the climate date is greater than or equal to the start date and less than or equal to the end date.

filter_comb <- big_comb %>% filter(clim_date >= start_date, clim_date <= end_date)

# Step 4: Convert the long/tall format climate data to wide format where you only have 1 row per individual with a column for each month and each climate variable that the specimen experienced over its year of interest (i.e. Tave_01 is the Average Temperature in January).


# Simplify to keep only necessary columns

filter_comb <- filter_comb %>% 
  select(-clim_year, -clim_date, -end_mo, -start_mo)

#Pivot the Data Frame to be wider and include new columns for every climate month paired with each climate variable.

wide_final_clim_subset <- filter_comb %>% pivot_wider(
    names_from = clim_month,
    values_from = c("CMD_mm", "CMI_mm", "chillDD", "heatDD", "coolDD", 
"growDD", "Eref", "FFD", "PAS", "ppt_mm", "RH", "srad", "tave", "tmax", "tmin")
  )


# Step 5: Write out the combined project csv... not going to write until I hopefully figure out the weird missing data

write.csv(wide_final_clim_subset, "Output_Files/climateNA_subset_data_tall.csv", row.names = FALSE)

#WUsing the full year (12 months) of data starting with the day and month of collection, but in the previous year, and ending with the date of collection. 

# METHODS: First, we need to calculate the date range. We will only include the month of collection in the year of collection if it was collected after the first week, so on the 8th or later. If it was collected before the 8th (1-7th), we do NOT include that month for the year of collection, and instead begin from the full month completed prior to the specimen's collection Additionally, if a specimen did not have a specific date of collection listed (n = 17), whcih displays as day=0 in the data frame, we do not feel confident in assuming that we should use said listed month as the end date, so the specimen will be treated the same as a specimen collected on the 1st-7th of the month. 


# Step 1: Identify the start and end month for the "climate year" for each specimen within the specimen df.These new columns should be in the clim_date format with year and month together (e.g. 202412 for December 2024).

# Step 1a: Identify the end month (this will then define the start month as 11 mo before). The newly coded end_mo column should contain the month of collection ( = month) if the day of collection column has a value >7. Otherwise, the end_mo column is assigned the month previous to collection (month -1). Special care was taken to address the cyclical nature of the calendar year, where Janaury is preceded by December.

specimens <- specimens %>% mutate(
  end_mo = if_else(day > 7, month, 
if_else(month == 1, 12, month - 1))) 

# Step 1b: Identify the start month and create its column; this will simply be the end month + 1 in order to capture the full 12 month (1 year) cycle. Once again, special care was taken to address the cyclical nature of the calendar year, where Janaury is preceded by December.


specimens <- specimens %>% mutate(start_mo = if_else(end_mo != 12, end_mo + 1, 1))

# Step 1c: Now there are two new columns in the specimens df, one each for start and end month. The year of collection must be appended to the month in order to get the start date and end date pertaining to each specimen that will narrow down the ClimateNA data. Special precautions were included to address the End Date scenario where a specimen collected before January 8th is assigned December as an end date, but the year must be listed (for both start and end date), as the year prior. Additionally, when constructing the Start Date character string, special conditions were added for specimens collected past December 7th, where, unlike any other collection date, the Start and End Date fall in the same year. 


#For End Date:
specimens <- specimens %>% mutate(end_date = if_else(end_mo == 12 & month == 1, paste0(year-1, str_pad(end_mo, 2, pad = "0")), paste0(year, str_pad(end_mo, 2, pad = "0"))), end_mo = as.integer(end_mo))

#For Start Date:
specimens <- specimens %>% mutate(start_date = if_else(month == 12 & end_mo == 12, paste0(year, str_pad(start_mo, 2, pad = "0")), paste0(year -1, str_pad(start_mo, 2, pad = "0"))), start_mo = as.integer(start_mo))

# Step 2: Join the specimens and clim dataframes to combine the newly adjusted CCH2 Data with the unfiltered ClimateNA data frame.

big_comb <- left_join(clim, specimens, by = "id")

# Step 3: To subset the climate data, keep only the rows for each specimen ID where the climate date is greater than or equal to the start date and less than or equal to the end date.

filter_comb <- big_comb %>% filter(clim_date >= start_date, clim_date <= end_date)

# Step 4: Convert the long/tall format climate data to wide format where you only have 1 row per individual with a column for each month and each climate variable that the specimen experienced over its year of interest (i.e. Tave_01 is the Average Temperature in January).


# Simplify to keep only necessary columns

filter_comb <- filter_comb %>% 
  select(-clim_year, -clim_date, -end_mo, -start_mo)

#Pivot the Data Frame to be wider and include new columns for every climate month paired with each climate variable.

wide_final_clim_subset <- filter_comb %>% pivot_wider(
    names_from = clim_month,
    values_from = c("CMD_mm", "CMI_mm", "chillDD", "heatDD", "coolDD", 
"growDD", "Eref", "FFD", "PAS", "ppt_mm", "RH", "srad", "tave", "tmax", "tmin")
  )


# Step 5: Write out the combined project csv... not going to write until I hopefully figure out the weird missing data
